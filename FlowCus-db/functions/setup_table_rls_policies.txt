CREATE OR REPLACE FUNCTION public.setup_table_rls_policies(table_name text)
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
    -- Check if table exists
    IF NOT EXISTS (
        SELECT 1 FROM information_schema.tables 
        WHERE table_schema = 'public' 
        AND table_name = $1
    ) THEN
        RAISE EXCEPTION 'Table "%" does not exist in public schema', $1;
    END IF;
    
    -- Enable RLS
    EXECUTE format('ALTER TABLE %I ENABLE ROW LEVEL SECURITY', table_name);
    
    -- Drop existing policies if they exist (to avoid conflicts)
    EXECUTE format('
        DROP POLICY IF EXISTS "appuser_select_all_%s" ON %I',
        table_name, table_name);
    EXECUTE format('
        DROP POLICY IF EXISTS "appuser_insert_all_%s" ON %I',
        table_name, table_name);
    EXECUTE format('
        DROP POLICY IF EXISTS "appuser_update_all_%s" ON %I',
        table_name, table_name);
    EXECUTE format('
        DROP POLICY IF EXISTS "appguest_select_all_%s" ON %I',
        table_name, table_name);
    
    -- Create new policies (same as before)
    EXECUTE format('
        CREATE POLICY "appuser_select_all_%s" 
        ON %I 
        FOR SELECT 
        TO appuser 
        USING (true)', 
        table_name, table_name);
        
    EXECUTE format('
        CREATE POLICY "appuser_insert_all_%s" 
        ON %I 
        FOR INSERT 
        TO appuser 
        WITH CHECK (true)', 
        table_name, table_name);
        
    EXECUTE format('
        CREATE POLICY "appuser_update_all_%s" 
        ON %I 
        FOR UPDATE 
        TO appuser 
        USING (true) 
        WITH CHECK (true)', 
        table_name, table_name);
    
    EXECUTE format('
        CREATE POLICY "appguest_select_all_%s" 
        ON %I 
        FOR SELECT 
        TO appguest 
        USING (true)', 
        table_name, table_name);
    
    RAISE NOTICE 'Successfully configured RLS and policies for table: %', table_name;
END;
$$;
