CREATE TABLE tasks (
    task_id SERIAL PRIMARY KEY,
    task_master_id INT REFERENCES task_master(id),
    title VARCHAR(200),
    description TEXT,
    priority INTEGER,
    created_on TIMESTAMP NOT NULL DEFAULT NOW(),
    created_by INT NOT NULL REFERENCES userlist(id),
    updated_on TIMESTAMP,
    updated_by INT REFERENCES userlist(id),
    start_time TIMESTAMP,
    end_time TIMESTAMP,
    duration_seconds INTEGER,
    is_deleted BOOLEAN DEFAULT FALSE
);
